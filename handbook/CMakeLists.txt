# Test harness for every code snippet in the handbook (built by CI on Windows with MSVC / clang-cl)
cmake_minimum_required(VERSION 3.21)
project(handbook LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HB_UTF8 "compile with /utf-8 (recommended in the handbook)" ON)
option(HB_SIM936 "simulate a Chinese Windows (code page 936) by /source-charset:.936" OFF)
if(MSVC)
  if(HB_UTF8)
    add_compile_options(/utf-8)
  endif()
  if(HB_SIM936)
    add_compile_options(/source-charset:.936)
  endif()
endif()
add_subdirectory(early)
add_subdirectory(late)
add_subdirectory(rt)
