            "texture;TextureShader;s7_texture.inc;_nm.tga" "tangent;TangentShader;s7_tangent.inc;_nm_tangent.tga")
  list(GET cfg 0 n)
  list(GET cfg 1 sh)

  target_compile_definitions(r_${n} PRIVATE SHADER=${sh} "SHADER_FILE=\"${f}\"" "NM_SUFFIX=\"${nm}\"")
endforeach()
